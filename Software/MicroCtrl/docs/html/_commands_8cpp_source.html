58Command Commands::Parse(String &input)
59{
60 input.trim();
61 input.toLowerCase();
62
63 log(input, "CMD::Parse:[" + String(_range) + "] Incoming Data: ");
64
65 int commandIndex = input.indexOf(" ");
66 String commandName = input.substring(0, commandIndex);
67 String commandValue = input.substring(commandIndex, input.length());
68 commandValue.trim();
69
70 log(commandValue.length(), "CMD::Parse: CommandValue lenght: ");
71
72 for (int itr = 0; itr < _range; itr++)
73 {
74 if (_list[itr].Name.equals(commandName))
75 {
76 if (commandValue.length() > 0)
77 {
78 _list[itr].Content = commandValue;
79 _list[itr].Value = commandValue.toFloat();
80 }
81 log(_list[itr]);
82 return _list[itr];
83 }
84 }
85
86 log("CMD::Parse: Command Validation failed");
87 return {-1, "Invalid", 0};
88}

61// Forward-Kinematics calculation with respect to given Posture
62Position Kinematic::fdKinematic(Posture pt){
63 tmm::Matrix<4,4> mat = armTMatrix(pt);
64 printMatrix(mat, "Arm Transformation Matrix");
65
66 Position ps {};
67 ps.X = mat[0][3];
68 ps.Y = mat[1][3];
69 ps.Z = mat[2][3];
70
71 ps.B = getDeg(atan2(-mat[2][0], sqrtf(powf(mat[0][0],2) + powf(mat[1][0],2))));
72 ps.A = getDeg(atan2(mat[1][0]/cos(getRad(ps.B)), mat[0][0]/cos(getRad(ps.B))));
73 ps.C = getDeg(atan2(mat[2][1]/cos(getRad(ps.B)), mat[2][2]/cos(getRad(ps.B))));
74
75 if(ps.B == (getDeg(pi)/2)){
76 ps.A = 0;
77 ps.C = getDeg(atan2(mat[0][1], mat[1][1]));
78 }
79 if((ps.B == -(getDeg(pi)/2))) {
80 ps.A = 0;
81 ps.C = getDeg(-atan2(mat[0][1], mat[2][2]));
82 }
83
84 return ps;
85}

87// Berechung der Positionsgebenden Winkelstellungen
88// d.h. th1, th2 und th3
89Posture Kinematic::ivPosition(Position ps){
90 Posture pt = {};
91 logln(ps, "ivPosition: Target Position");
92
93 // Berechnung von theata 1
94 float jt1 = atan2(ps.Y, ps.X);
95 pt.Jt1 = getDeg(jt1);
96
97 logft(jt1, "jt1");
98 logft(pt.Jt1, "pt.jt1");
99
100 // Ermittlung von theta2 durch Anwendung der Linear Kombination von Sinus
101 // und Kosinus wellen
102 // ie. A*cos(x) + B*sin(x) = C*cos(x + phi) with
103 // C = sgn(A).Sqrt(A^2 + B^2) ie. sgn(A) = (A/|A|)
104 // phi = arctan(-B/A)
105 //
106 // Berechnung von theta 2
107 float A = 2*ps.Z*dhPar2.link;
108 float B = -2*ps.X*cos(jt1)*dhPar2.link-2*ps.Y*sin(jt1)*dhPar2.link+2*dhPar1.link*dhPar2.link;
109 float C = sign(A)*sqrt(powf(A,2)+powf(B,2));
110 float D = -powf(dhPar3.link,2)+(powf(ps.X,2)*powf(cos(jt1),2))+(powf(ps.Y,2)*powf(sin(jt1),2))
111 +(2*ps.X*cos(jt1)*ps.Y*sin(jt1))-(2*ps.X*cos(jt1)*dhPar1.link)-(2*ps.Y*sin(jt1)*dhPar1.link)
112 +powf(dhPar2.link,2)+powf(dhPar1.link,2)+powf(ps.Z,2);
113 float Phi = atan2(-B,A);
114 // d.h
115 float jt2 = atan2(sqrt(1-powf(D/C,2)),(D/C)) - Phi;
116 float jt2_neg = atan2(-sqrt(1-powf(D/C,2)),(D/C)) - Phi;
117
118 pt.Jt2 = getDeg(jt2);
119
120 logft(jt2, "jt2");
121 logft(pt.Jt2, "pt.jt2");
122
123 // Berechnung von theta 3:
124 float jt3 = atan2(-ps.X*cos(jt1)-ps.Y*sin(jt1)-dhPar2.link*sin(jt2)+dhPar1.link, (ps.Z-dhPar2.link*cos(jt2)))-jt2;
125 pt.Jt3 = getDeg(jt3);
126
127 logft(jt3, "jt3");
128 logft(pt.Jt3, "pt.jt3");
129
130 return pt;
131}

143// Ermittlung der Orientierung/Gelenkstellung
144// d.h. th4, th5 und th6
145Posture Kinematic::ivOrientation(Posture pt, Position ps){
146 // D-H Parameter Zuweisen
147 dhPar1.theta = pt.Jt1;
148 dhPar2.theta = pt.Jt2;
149 dhPar3.theta = pt.Jt3;
150 // Achsen-Transformations Matrizen Ermittlen
151 tMat01 = jointTMatrix(dhPar1);
152 tMat12 = jointTMatrix(dhPar2);
153 tMat23 = jointTMatrix(dhPar3);
154 // Positins-Transformations Matrizen Ermittlen
155 tMat03 = tMat01*tMat12*tMat23;
156 tmm::Matrix<3,3> rMat03 = read3x3(tMat03);
157
158 // Ziel Rotations Matrix bauen
159 DhParam dhp = {ps.A, ps.B, ps.C,0,0};
160 tmm::Matrix<4,4> tMat06 = jointTMatrix(dhp);
161 tmm::Matrix<3,3> rMat06 = read3x3(tMat06);
162
163 // Orientierungs Winkeln
164 tmm::Matrix<3,3> rMat36 = rMat03.inverse() * rMat06;
165
166 // Berechnung von theta 4
167 pt.Jt4 = getDeg(atan2(-rMat36[0][2],rMat36[1][2]));
168
169 // Berechnung von theta 5
170 pt.Jt5 = getDeg(atan2(sqrt(powf(rMat36[0][2],2)+powf(rMat36[1][2],2)),rMat36[2][2]));
171
172 // Berechnung von theta 6
173 pt.Jt6 = getDeg(atan2(rMat36[2][0],rMat36[2][1]));
174
175 return pt;
176}

178// Inverse-Kinematics calculation with respect to given Position
179Posture Kinematic::ivKinematic(Position p6){
180 Posture pt = {};
181
182 // Berechnung der Positionsgebenden Gelenkstellung d.h. th1, th2 und th3
183 //
184 // TMat = |RotationsMatrix(3x3) Verschiebung(3,1)|
185 // | 0 1 |
186 //
187 // TMat03 = TMat1*TMat2*TMat3
188 // TMat03 =...
189 // |-sin(th1) -cos(th1)*sin(th2+th3) cos(th1)*cos(th2+th3) cos(th1)*(a(3)*cos(th2+th3)+a(2)*cos(th2)+a(1))|
190 // | cos(th1) -sin(th1)*sin(th2+th3) sin(th1)*cos(th2+th3) sin(th1)*(a(3)*cos(th2+th3)+a(2)*cos(th2)+a(1))|
191 // | 0 cos(th2+th3) cos(th2+th3) a(3)*sin(th2+th3)+a(2)*sin(th2) |
192 // | 0 0 0 1 |
193 //
194 pt = ivPosition(p6);
195
196
197 // Berechnung der Orientierungsgebenden Gelenkstellung d.h. th4, th5 und th6
198 // RMat0_6 = RMat0_3 * RMat3_6
199 // (RMat0_3)^-1 * RMat0_6 = (RMat0_3)^-1 * RMat0_3 * RMat3_6
200 // d.h.
201 // RMat3_6 = (RMat0_3)^-1 * RMat0_6
202 //
203 // mit TMat0_6 = |RotationsMatrix(3x3) Verschiebung(3x1)|
204 // | 0(1x3) 1 |
205 //
206 // TMat36 = TMat4*TMat5*TMat6
207 // TMat36 =...
208 // |-cos(th4)*cos(th6)-sin(th4)*cos(th5)*sin(th6) cos(th4)*sin(th6)-sin(th4)*cos(th5)*cos(th6) -sin(th4)*sin(th5) -a(6)*cos(th4)*cos(th6)-a(6)*sin(th4)*cos(th5)*sin(6)-a(5)*sin(th4)*sin(th5)+a(4)*cos(4)|
209 // |-sin(th4)*cos(th6)+cos(th4)*cos(th5)*sin(th6) sin(th4)*sin(th6)+cos(th4)*cos(th5)*cos(th6) cos(th4)*sin(th5) -a(6)*sin(th4)*cos(th6)-a(6)*cos(th4)*cos(th5)*sin(6)-a(5)*cos(th4)*sin(th5)+a(4)*sin(4)|
210 // | sin(th5)*sin(th6) sin(th5)*cos(th6) -cos(th5) a(6)*sin(th5)*sin(th6)-a(5)*cos(th5) |
211 // | 0 0 0 0 |
212 //
213 // Ermittlung der Manipulator Transformations-Matrix entsprechend der
214 // gegebenen Positions-Winkelwerten
215 pt = ivOrientation(pt, p6);
216
217 // Ermittlung der Positions Difference zwischen Endeffektor und Gelenk 5
218 // unterberucksichtigung der Nickung in Acshe 2, 3 oder 5
219 if(pt.Jt2 || pt.Jt3 || pt.Jt5){
220 // Gelenk 5 entlang der Orientierungs Achse verschieben
221 // Matrix4x4 = tmat06v*TransFK(0,90,0,0,0)*TransFK(0,0,0,2*a(5),0);
222 //
223 // NOTE: Validiere beide Matrizen dhParOpp und tMatDis zusammen
224 // getragen werden kann.
225 DhParam dhParOpp = {0.000, 90.000, 0.000, 0.000, 0.000};
226 DhParam dhParDis = {0.000, 0.000, 0.000, 2*dhPar5.link, 0.000};
227 tmm::Matrix<4,4> tMat06 = armTMatrix(pt);
228 tmm::Matrix<4,4> tMatOpp = jointTMatrix(dhParOpp);
229 tmm::Matrix<4,4> tMatDis = jointTMatrix(dhParDis);
230
231 // Neuer Ziel Position Berechnen
232 tmm::Matrix<4,4> tMat06Corr = tMat06*tMatOpp*tMatDis;
233
234 // Berechnung der Positionsgebenden Gelenkstellung d.h. th1, th2 und th3
235 // unter Berücksichtigung von Achse 56
236 Position p6Corr =
237 {tMat06Corr[0][3], tMat06Corr[1][3], tMat06Corr[2][3], p6.A, p6.B, p6.C};
238 pt = ivPosition(p6Corr);
239
241 pt = ivOrientation(pt, p6Corr);
242 }
243
244
245 logln(p6, "ivKinematic: Target Position");
246 return pt;
247}

249// Returns Joint-Transformation Matrix with respect
250// to given of Dh-Parameters
251tmm::Matrix<4,4> Kinematic::jointTMatrix(DhParam dhPar){
252 // Denavit-Hartenberg-Transformationmatrix (a alpha beta d theta)
253 //
254 //TMat = [ cos(beta)*cos(gamma) -cos(alpha)*sin(gamma)+sin(alpha)*sin(beta)*cos(gamma) sin(alpha)*sin(gamma)+cos(alpha)*sin(beta)*cos(gamma) a*cos(gamma);
255 // cos(beta)*sin(gamma) cos(alpha)*cos(gamma)+sin(alpha)*sin(beta)*sin(gamma) -sin(alpha)*cos(gamma)+cos(alpha)*sin(beta)*sin(gamma) a*sin(gamma);
256 // -sin(beta) sin(alpha)*cos(beta) cos(alpha)*cos(beta) d ;
257 // 0 0 0 1 ];
258 //
259 //TMat = [m[0][0], m[0][1], m[0][2], m[0][3];
260 // m[1][0], m[1][1], m[1][2], m[1][3];
261 // m[2][0], m[2][1], m[2][2], m[2][3];
262 // m[3][0], m[3][1], m[3][2], m[3][3];]
263 //
264
265 tmm::Matrix<4,4> mat = {};
266
267 // Matrix m11 bis m41
268 mat.printTo(Serial);
269 mat[0][0] = cos(getRad(dhPar.beta))*cos(getRad(dhPar.theta));
270 mat[1][0] = cos(getRad(dhPar.beta))*sin(getRad(dhPar.theta));
271 mat[2][0] = -sin(getRad(dhPar.beta));
272 mat[3][0] = 0;
273 // Matrix m12 bis m42
274 mat[0][1] = -cos(getRad(dhPar.alpha))*sin(getRad(dhPar.theta))+sin(getRad(dhPar.alpha))*sin(getRad(dhPar.beta))*cos(getRad(dhPar.theta));
275 mat[1][1] = cos(getRad(dhPar.alpha))*cos(getRad(dhPar.theta))+sin(getRad(dhPar.alpha))*sin(getRad(dhPar.beta))*sin(getRad(dhPar.theta));
276 mat[2][1] = sin(getRad(dhPar.alpha))*cos(getRad(dhPar.beta));
277 mat[3][1] = 0;
278 // Matrix m13 bis m43
279 mat[0][2] = sin(getRad(dhPar.alpha))*sin(getRad(dhPar.theta))+cos(getRad(dhPar.alpha))*sin(getRad(dhPar.beta))*cos(getRad(dhPar.theta));
280 mat[1][2] = -sin(getRad(dhPar.alpha))*cos(getRad(dhPar.theta))+cos(getRad(dhPar.alpha))*sin(getRad(dhPar.beta))*sin(getRad(dhPar.theta));
281 mat[2][2] = cos(getRad(dhPar.alpha))*cos(getRad(dhPar.beta));
282 mat[3][2] = 0;
283 // Matrix m14 bis m44
284 mat[0][3] = dhPar.link*cos(getRad(dhPar.theta));
285 mat[1][3] = dhPar.link*sin(getRad(dhPar.theta));
286 mat[2][3] = dhPar.disp;
287 mat[3][3] = 1;
288
289 return mat;
290}

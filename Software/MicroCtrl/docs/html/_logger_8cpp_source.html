86 String ptEncode(float val){
87 String jt = "";
88
89 String jts = String(round(abs(val)));
90
91 if(jts.length() <= 1){
92 jts = "00" + jts;
93 }
94 if(jts.length() <= 2){
95 jts = "0" + jts;
96 }
97
98 if(val > 0){
99 jt = "+" + jts;
100 }
101 else {
102 jt = "-" + jts;
103 }
104 return jt;
105 }

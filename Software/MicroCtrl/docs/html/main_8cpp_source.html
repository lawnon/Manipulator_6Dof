84// Execute Program Here
85void loop()
86{
87 CycleCount++;
88
89 if (!digitalRead(PinSW))
90 {
91 log("PinSW Pressed");
92 }
93
94 // Get Inputs from Serial Port and Process
95 if (Serial.available() > 0)
96 {
97 RobArm.Decode(Serial.readString());
98 }
99
100 if (RotDetected >= 1)
101 {
102 int step = 0;
103 if (RotDirection >= 1)
104 {
105 step = 1 * RobArm.Commands.GetParam(Commands::RotStep);
106 }
107 else
108 {
109 step = -1 * RobArm.Commands.GetParam(Commands::RotStep);
110 }
111 // log(step, "Rotation-Step");
112 switch (RobArm.Commands.GetParam(Commands::Rotate))
113 {
114 case Commands::Jt1:
115 RobArm.GotoDeg(RobArm.Joint1, RobArm.Joint1.Read() + step);
116 break;
117 case Commands::Jt2:
118 RobArm.GotoDeg(RobArm.Joint2, RobArm.Joint2.Read() + step);
119 break;
120 case Commands::Jt3:
121 RobArm.GotoDeg(RobArm.Joint3, RobArm.Joint3.Read() + step);
122 break;
123 case Commands::Jt4:
124 RobArm.GotoDeg(RobArm.Joint4, RobArm.Joint4.Read() + step);
125 break;
126 case Commands::Jt5:
127 RobArm.GotoDeg(RobArm.Joint5, RobArm.Joint5.Read() + step);
128 break;
129 case Commands::Jt6:
130 RobArm.GotoDeg(RobArm.Joint4, RobArm.Joint6.Read() + step);
131 break;
132 }
133 RotDetected = -1;
134 }
135
136 // Loop Defined Positions
137 RobArm.LoopPositions();
138
139 if (Serial2.available() > 0)
140 {
141 RobArm.Drive(RobArm.PosFactory.DecodePosture(Serial2.readString()));
142 }
143
144 if ((CycleCount % 50000) == 0 && false)
145 {
146 AnalogValue = analogRead(AnalogIn);
147 MeasuredVoltage = (AnalogValue * RefVoltage) / 1024;
148 BatteryVoltage = (MeasuredVoltage * (Res1 + Res2)) / Res2;
149
150 log(CycleCount, "Operation Counter: ");
151 log(BatteryVoltage, "Estimated Battery Voltage: ");
152
153 if (BatteryVoltage <= RobArm.Commands.GetParam(Commands::Voltage))
154 {
155 logToSerial("sleep");
156 }
157 else
158 {
159 logToSerial(Posture{RobArm.Joint1.Read(),
160 RobArm.Joint2.Read(),
161 RobArm.Joint3.Read(),
162 RobArm.Joint4.Read(),
163 RobArm.Joint5.Read(),
164 RobArm.Joint6.Read()});
165 }
166 }

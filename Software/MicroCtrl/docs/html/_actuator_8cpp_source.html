85int32 Actuator::Activate(int8 ena, uint16 tof)
86{
87 if(ena >= ON)
88 {
89 _stepper->enable();
90 _state = SetBit16(_state, State::Enabled);
91 _timeOfDelay = tof;
92 //log(GetBit16(_state, State::InPosition), "Act:Inposition: ");
93 //log(_targetSteps, "Act:target: ");
94 //log(_actualSteps, "Act:Actual: ");
95 if(GetBit16(_state, State::InPosition) <= OFF)
96 {
97 if(_stepper->nextAction() >= ON)
98 {
99 _state = SetBit16(_state, State::InMotion);
100 _state = ResetBit16(_state, State::Stopped);
101
102 _actualSteps = (_dir >= ON) ?
103 _previousSteps + _stepper->getStepsCompleted():
104 _previousSteps - _stepper->getStepsCompleted();
105 }
106 else
107 {
108 if (abs(_targetSteps -_actualSteps) <= Accuracy)
109 {
110 _state = SetBit16(_state, State::InPosition);
111 }
112 }
113 return _stepper->nextAction();
114 }
115 }
116 else
117 {
118 if(GetBit16(_state, State::Enabled) >= ON)
119 {
120 _timeOfDelay--;
121 if(_timeOfDelay <= 0)
122 {
123 _stepper->disable();
124 _state = ResetBit16(_state, State::Enabled);
125 _state = ResetBit16(_state, State::InMotion);
126 _state = SetBit16(_state, State::Stopped);
127 }
128 return ON;
129 }
130 }
131 return OFF;
132}

135int32 Actuator::Write(int32 target)
136{
137 if (target < _min)
138 {
139 target = _min;
140 }
141
142 if (target > _max)
143 {
144 target = _max;
145 }
146 int16 actpos = calcPosition(_actualSteps);
147 if(target >= actpos)
148 {
149 target = target - actpos;
150 }
151 else
152 {
153 target = actpos - target;
154 }
155
156 _previousSteps = _actualSteps;
157 _targetSteps = calcSteps(target);;
158
159 // Delta ungleich Null Validierung
160 if(_targetSteps == _actualSteps)
161 {
162 _state = SetBit16(_state, State::Aborted);
163 return NOK;
164 }
165
166 // Fahrtrichtungs Validierung
167 _dir = (_targetSteps >= _actualSteps) ? 1 : -1;
168 _stepper->startMove(_targetSteps*_dir);
169 log(_targetSteps, "Act:target: ");
170 log(_actualSteps, "Act:Actual: ");
171 log(_dir, "Act:Write dir ");
172 log(_stepper->getDirection(), "Act:Write: Stepper Dir: ");
173 if(_dir != _stepper->getDirection())
174 {
175 log(_id, "Act:Write Refernce Actor; ");
176 _state = SetBit16(_state, State::Aborted);
177 _state = ResetBit16(_state, State::Referenced);
178 _stepper->startMove(0);
179 return NOK;
180 }
181
182 log(_targetSteps, "Ziel Schritte: ");
183 _state = ResetBit16(_state, State::InPosition);
184
185
186 while (Activate(ON)>OK);
187 return _targetSteps;
188}

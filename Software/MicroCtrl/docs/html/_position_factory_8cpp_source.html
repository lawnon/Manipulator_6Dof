96 input.substring(0, delimeter1).toFloat(),
97 input.substring(delimeter1 + 1, delimeter2).toFloat(),
98 input.substring(delimeter2 + 1, delimeter3).toFloat(),

154int& PositionFactory::AddPosition(String content, PosState state, char filter = ' '){
155 /* Expected String := "Position <identifier> <x>, <y>, <z>, <a>, <b>, <c>" */
156
157 int result = -1;
158 int delimeter = content.indexOf(filter);
159 String identifier = content.substring(0, fmax(content.length(), IDENTIFIER_LEN));
160 String params = "";
161
162 if(delimeter >= 1){
163 identifier = content.substring(0, fmin(delimeter, IDENTIFIER_LEN));
164 params = content.substring(delimeter + 1, content.length());
165 }
166
167 //log("PF::AddPosition: Content: " + content);
168 //log("PF::AddPosition: Identifier: " + identifier);
169 //log("PF::AddPosition: Parameters: " + params);
170 //log(delimeter, "PF::AddPosition: Delimeter ");
171
172 if(identifier.length() <= 0 ||
173 identifier.equals(_robFrame->Commands.Get(Commands::Here).Name) ||
174 identifier.equals(_robFrame->Commands.Get(Commands::Joints).Name)){
175 log("PF::AddPosition: Invalid Identifier");
176 return result;
177 }
178
179 char cIdentifier[IDENTIFIER_LEN];
180 identifier.toCharArray(cIdentifier, IDENTIFIER_LEN);
181
182 for(int itr = 0; itr < _definedCount; itr++){
183 if(strcmp(_data[itr].Identifier, cIdentifier) == 0){
184 result = itr;
185 Update(itr, params, state);
186 log(_data[itr], state);
187 return result;
188 }
189 }
190
191 if(RangeCheck() <= 0){
192 log("PF::AddPos: Range Exceeded");
193 return result;
194 }
195
196 strcpy(_data[_definedCount].Identifier, cIdentifier);
197 Update(_definedCount, params, state);
198
199 log(_data[_definedCount], state);
200 _definedCount++;
201
202 result = _definedCount - 1;
203 return result;
204}

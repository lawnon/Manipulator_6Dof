99Position RobFrame::Interpolate(Position& origin,
100 Position& dest,
101 int range,
102 int step = 1){
103 return Position {
104 ((dest.X - origin.X) > range) ? origin.X + step*((dest.X - origin.X)/abs(dest.X - origin.X)) : dest.X,
105 ((dest.Y - origin.Y) > range) ? origin.Y + step*((dest.Y - origin.Y)/abs(dest.Y - origin.Y)) : dest.Y,
106 ((dest.Z - origin.Z) > range) ? origin.Z + step*((dest.Z - origin.Z)/abs(dest.Z - origin.Z)) : dest.Z,
107 ((dest.A - origin.A) > range) ? origin.A + step*((dest.A - origin.A)/abs(dest.A - origin.A)) : dest.A,
108 ((dest.B - origin.B) > range) ? origin.B + step*((dest.B - origin.B)/abs(dest.B - origin.B)) : dest.B,
109 ((dest.C - origin.C) > range) ? origin.C + step*((dest.C - origin.C)/abs(dest.C - origin.C)) : dest.C
110 };

114PosState RobFrame::Drive(Posture pt){
115 log("Drive Data: " + String(pt.Jt1) +
116 " | " + String(pt.Jt2) +
117 " | " + String(pt.Jt3) +
118 " | " + String(pt.Jt4) +
119 " | " + String(pt.Jt5) +
120 " | " + String(pt.Jt6));
121
122 if(pt.Jt1 == 0 && pt.Jt2 == 0 && pt.Jt3 == 0
123 && pt.Jt4 == 0 && pt.Jt5 == 0 && pt.Jt6 == 0){
124 return;
125 }
126
127 int origin1 = Joint1.Read();
128 int origin2 = Joint2.Read();
129 int origin3 = Joint3.Read();
130 int origin4 = Joint4.Read();
131 int origin5 = Joint5.Read();
132 int origin6 = Joint6.Read();
133
134 int delta1 = Joint1.Delta(pt.Jt1);
135 int delta2 = Joint2.Delta(pt.Jt2);
136 int delta3 = Joint3.Delta(pt.Jt3);
137 int delta4 = Joint4.Delta(pt.Jt4);
138 int delta5 = Joint5.Delta(pt.Jt5);
139 int delta6 = Joint6.Delta(pt.Jt6);
140 log("Drive Delta: " + String(delta1) +
141 " | " + String(delta2) +
142 " | " + String(delta3) +
143 " | " + String(delta4) +
144 " | " + String(delta5) +
145 " | " + String(delta6));
146
147 // Get max delta
148 int delta = (abs(delta1) >= abs(delta2)) ? abs(delta1) : abs(delta2);
149 delta = (abs(delta) >= abs(delta3)) ? abs(delta) : abs(delta3);
150 delta = (abs(delta) >= abs(delta4)) ? abs(delta) : abs(delta4);
151 delta = (abs(delta) >= abs(delta5)) ? abs(delta) : abs(delta5);
152 delta = (abs(delta) >= abs(delta6)) ? abs(delta) : abs(delta6);
153
154 // Iterate Through delta Range and Increment
155 for(int itr = 1; itr <= delta; itr++){
156 //log(itr, "iterator");
157 Increment(Joint1, origin1, pt.Jt1, itr);
158 Increment(Joint2, origin2, pt.Jt2, itr);
159 Increment(Joint3, origin3, pt.Jt3, itr);
160 Increment(Joint4, origin4, pt.Jt4, itr);
161 Increment(Joint5, origin5, pt.Jt5, itr);
162 Increment(Joint6, origin6, pt.Jt6, itr);
163 // Delay with respect to Speed
164 Commands.Delay();
165 }
166 return PosState::Defined;
167}

253void RobFrame::Decode(String data) {
254 // Eingangs Daten lesen

256
257 switch (inputCmd.Id) {
258 case Commands::World:
259 log("World Command not yet Implemented");
260 break;
261
262 case Commands::Jt1:
263 GotoDeg(Joint1, inputCmd.Value);
264 break;
265
266 case Commands::Jt2:
267 GotoDeg(Joint2, inputCmd.Value);
268 break;
269
270 case Commands::Jt3:
271 GotoDeg(Joint3, inputCmd.Value);
272 break;
273
274 case Commands::Jt4:
275 GotoDeg(Joint4, inputCmd.Value);
276 break;
277
278 case Commands::Joints:
279 // Get Angular Positoin of each Joint
280 log(Posture {
281 Joint1.Read(),
282 Joint2.Read(),
283 Joint3.Read(),
284 Joint4.Read(),
285 Joint5.Read(),
286 Joint6.Read()
287 });
288 break;
289
290 case Commands::Here:
291 // Get Carthesian Koordinates
292 log(Kinematics.fdKinematic(Posture {
293 Joint1.Read(),
294 Joint2.Read(),
295 Joint3.Read(),
296 Joint4.Read(),
297 Joint5.Read(),
298 Joint6.Read()
299 }));
300 break;
301
302 case Commands::GoHome:
303 Drive(Posture {PtHome.Jt1, PtHome.Jt2, PtHome.Jt3, PtHome.Jt4, PtHome.Jt5, PtHome.Jt6});
304 break;
305
306 case Commands::Drive:
307 Drive(PosFactory.DecodePosition(inputCmd.Content));
308 break;
309
310 case Commands::JDrive:
311 Drive(PosFactory.DecodePosture(inputCmd.Content));
312 break;
313
314 case Commands::LDrive:
315 LDrive(PosFactory.DecodePosition(inputCmd.Content));
316 break;
317
318 case Commands::Position:
319 PosFactory.AddPosition(inputCmd.Content, PosState::Position);
320 break;
321
322 case Commands::Posture:
323 PosFactory.AddPosition(inputCmd.Content, PosState::Posture);
324 break;

326 case Commands::ListPositions:
327 PosFactory.LogPositions(PosState::Position);
328 break;
329
330 case Commands::ListPostures:
331 PosFactory.LogPositions(PosState::Posture);
332 break;
333
334 case Commands::ListAllPositions:

338 Commands.List();
